\section{Lemmas from field theory}

\begin{lemma} \label{lem:fixed field of cyclic subgroup}
\lean{fixed_field_of_cyclic_subgroup}
\leanok
Let $K/F$ be a finite Galois extension of fields. Then for any $g \in \Gal(K/F)$
and $x \in K$, if $g(x) = x$, then $x$ belongs to the fixed field of the cyclic subgroup of $\Gal(K/F)$ generated by $x$.
\end{lemma}
\begin{proof}
\leanok
\end{proof}

\begin{lemma} \label{lem:p-power extensions}
\lean{X_pow_sub_C_irreducible_iff_of_prime_pow}
\mathlibok
Let $F$ be a field.
Let $p$ be an odd prime.
Then for $a \in F$, $a$ is not a $p$-th power
if and only if for every positive integer $n$, the polynomial $x^{p^n}-a$ is irreducible over $F$.
(Note that this fails for $p=2$, for example $x^4+4 = (x^2-2x+2)(x^2+2x+2)$.)
\end{lemma}
\begin{proof}
\mathlibok
In Mathlib.
\end{proof}

\begin{proposition} \label{prop:artin schreier extensions}
\lean{cyclic_char_p_as_artin_schreier}
\leanok
Let $F$ be a field of prime characteristic $p$.
Let $K/F$ be a Galois extension of degree $p$.
Then there exists $z \in K$ with minimal polynomial over $K$ of the form $x^p-x-a$ for some $a \in F$.
\end{proposition}
\begin{proof}
\uses{lem:fixed field of cyclic subgroup}
\leanok
Let $g \in \Gal(K/F)$ be a generator.
Since $K/F$ is separable, the trace map $K \to F$ is nonzero;
we can thus choose $y \in K$ with 
\[
1 = \Trace_{K/F}(y) = y + g(y) + \cdots + g^{p-1}(y).
\]
Set
\[
z := g(y) + 2g^2(y) + \cdots + (p-1) g^{p-1}(y);
\]
then $z - g(z) = 1$, and so $z^p-z$ is fixed by $g$. By Lemma~\ref{lem:fixed field of cyclic subgroup}, $z^p-z \in F$. Set $a := z^p - z$;
the isomorphism in question now carries $z$ to the class of $x$.
Since the polynomials $x^p-x$ and the minimal polynomial of $z$ are monic of the same degree $p$
(the latter because it must divide $[K:F] = p$ and cannot equal $1$) and the latter divides the former, they coincide.
\end{proof}

\begin{lemma} \label{lem:nonsquare in quadratic extension}
\lean{nonsquare_in_quadratic_extension}
\leanok
Let $K/F$ be a quadratic extension of fields of characteristic $\neq 2$.
Then there exists $a \in F$ which is not a square.
\end{lemma}
\begin{proof}
\leanok
Since $F$ is not of characteristic 2, $K/F$ is necessarily separable.
We may thus apply Kummer theory to write $K \cong F[x]/(x^2-a)$ for some $a \in F$.
This gives the desired element.
\end{proof}
